cmake_minimum_required(VERSION 3.16)
project(esdf_rviz_plugin)
find_package(ament_cmake REQUIRED)
find_package(rviz_default_plugins REQUIRED)
find_package(rviz_rendering REQUIRED)
find_package(pluginlib REQUIRED)
find_package(OpenCV REQUIRED COMPONENTS core imgproc)
add_library(${PROJECT_NAME} SHARED src/heatmap.cpp)
target_compile_features(${PROJECT_NAME} PUBLIC cxx_std_17)
target_include_directories(${PROJECT_NAME} SYSTEM PRIVATE ${OpenCV_INCLUDE_DIRS})
target_link_libraries(${PROJECT_NAME} ${OpenCV_LIBS})
ament_target_dependencies(${PROJECT_NAME} rviz_default_plugins rviz_rendering pluginlib)
pluginlib_export_plugin_description_file(rviz_common plugins.xml)
install(TARGETS ${PROJECT_NAME} LIBRARY DESTINATION lib)
ament_package()
